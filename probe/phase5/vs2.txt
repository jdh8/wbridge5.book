# Seats: 2C = (2♣) by East, South direct.  2C-P-2D = (2♣) P (2♦), South 4th.
# Default E-W: 2♣ and 2♦ strong artificial.  EO2faible=0: 2♦/2♥/2♠ natural
# strong (EO2Kfort then ignored).  No existing data covers these auctions.
# Every prefix call is scripted, so keep rate is 100%.

# --- (2♣) direct: broad, then weak 6-card majors, clubs, two-suiters ---
2C --hcp 6-18 -n 400 --seed 1 -o probe/data/p5/vs2/2C.jsonl
2C --hcp 0-11 --S 6- --seed 1 -o probe/data/p5/vs2/2C~S6low.jsonl
2C --hcp 4-14 --C 6- --seed 1 -o probe/data/p5/vs2/2C~C6.jsonl
2C --D 5- --C 5- --hcp 0-12 --seed 1 -o probe/data/p5/vs2/2C~mm55.jsonl
2C --H 5- --S 5- --hcp 0-10 --seed 1 -o probe/data/p5/vs2/2C~MM55low.jsonl
# --- (2♣) P (2♦): sandwich over the waiting response ---
2C-P-2D --hcp 6-18 -n 400 --seed 1 -o probe/data/p5/vs2/2CP2D.jsonl
2C-P-2D --D 5- --hcp 6-16 --seed 1 -o probe/data/p5/vs2/2CP2D~D5.jsonl
2C-P-2D --H 5- --S 5- --seed 1 -o probe/data/p5/vs2/2CP2D~MM55.jsonl
# --- (2♦) strong artificial GF (default): broad, weak hearts, minors ---
2D --hcp 6-18 -n 400 --seed 1 -o probe/data/p5/vs2/2D.jsonl
2D --hcp 0-11 --H 6- --seed 1 -o probe/data/p5/vs2/2D~H6low.jsonl
2D --D 5- --C 5- --hcp 0-14 --seed 1 -o probe/data/p5/vs2/2D~mm55.jsonl
# --- (2♦) P (2♥ no ace) ---
2D-P-2H --hcp 6-18 --seed 1 -o probe/data/p5/vs2/2DP2H.jsonl
# --- Natural strong twos (EO2faible=0), direct ---
2D --hcp 6-20 --ini EO2faible=0 --seed 1 -o probe/data/p5/vs2/n2D.jsonl
2H --hcp 6-20 -n 400 --ini EO2faible=0 --seed 1 -o probe/data/p5/vs2/n2H.jsonl
2H --S 4- --hcp 10-20 --ini EO2faible=0 --seed 1 -o probe/data/p5/vs2/n2H~S4.jsonl
2H --D 5- --C 5- --hcp 4-16 --ini EO2faible=0 --seed 1 -o probe/data/p5/vs2/n2H~mm55.jsonl
2S --hcp 6-20 --ini EO2faible=0 --seed 1 -o probe/data/p5/vs2/n2S.jsonl
2S --H 5- --hcp 8-20 --ini EO2faible=0 --seed 1 -o probe/data/p5/vs2/n2S~H5.jsonl
# --- Natural strong (2♥): after the 2NT negative, and in balancing seat ---
2H-P-2N --hcp 6-18 --ini EO2faible=0 --seed 1 -o probe/data/p5/vs2/n2HP2N.jsonl
